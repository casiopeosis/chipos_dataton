""" Semáforo de Inversión Social - CDMX Prototipo inicial (v0) del pipeline de análisis. QUÉ HACE ESTE PROTOTIPO ------------------------ Consolida las 5 ediciones históricas de DENUE ya depuradas en data/ (nov-2022, nov-2023, nov-2024, may-2025, may-2026), agregadas por alcaldía, y con ellas: 1. Mide la evolución real de la oferta (conteo de establecimientos) por alcaldía y categoría a lo largo de ~3.5 años. 2. Ajusta una tendencia lineal (mínimos cuadrados) oferta vs. tiempo, con intervalo de predicción para 1, 3 y 5 años. 3. Valida el modelo retrospectivamente: entrena con las 3 primeras ediciones (2022-2024) y compara la predicción contra lo observado en 2025 y 2026 (tal como pide el reto: "predecir 2020 usando 2010-2015"). 4. Normaliza la oferta actual por población (Censo 2020, INEGI) para tener una primera aproximación de oferta per cápita. 5. Clasifica cada alcaldía-categoría en un semáforo (rojo/amarillo/verde) combinando nivel de oferta per cápita y tendencia de crecimiento. QUÉ NO HACE TODAVÍA (limitación explícita, no implícita) --------------------------------------------------------- El README del proyecto pide cruzar la oferta con la DEMANDA real: cambios demográficos por AGEB (Censo 2010/2020 a nivel manzana), áreas verdes, PILARES, instalaciones deportivas de Datos Abiertos CDMX, directorio INAPAM y uso de suelo. Ninguna de esas fuentes está todavía en data/ — sólo DENUE. Por eso el semáforo de este v0 es un proxy de SOLO OFERTA (saturación / crecimiento de establecimientos), no todavía de demanda insatisfecha. La población usada aquí es un total estático 2020 por alcaldía (para tener algo per cápita ya), no una serie temporal de crecimiento poblacional. Esto se señala también en el dashboard para no engañar al usuario final. Próximo paso obvio: sustituir POBLACION_2020_ALCALDIA por el cruce real Censo 2010 vs 2020 por AGEB, y agregar espacio público / PILARES / INAPAM para construir el índice de demanda insatisfecha que sí pide el reto. """ import json from pathlib import Path import numpy as np import pandas as pd DATA_DIR = Path(__file__).resolve().parent.parent / "data" # (carpeta, archivo csv, año elapsed en meses desde la primera edición) EDITIONS = [ ("NOVIEMBRE 2022 CDMX", "denue_cdmx_11_2022_depurado.csv"), ("NOVIEMBRE 2023 CDMX", "denue_cdmx_11_2023_depurado.csv"), ("NOVIEMBRE 2024 CDMX", "denue_cdmx_11_2024_depurado.csv"), ("MAYO 2025 CDMX", "denue_cdmx_05_2025_depurado.csv"), ("MAYO 2026 CDMX", "denue_cdmx_05_2026_depurado.csv"), ] # Población total, Censo de Población y Vivienda 2020 (INEGI, comunicado # 98/21). Cifra estática de referencia -- ver limitación en el docstring. POBLACION_2020_ALCALDIA = { "Iztapalapa": 1835486, "Gustavo A. Madero": 1173351, "Álvaro Obregón": 759137, "Coyoacán": 614447, "Tlalpan": 699928, "Cuauhtémoc": 545884, "Azcapotzalco": 432205, "Iztacalco": 404695, "Benito Juárez": 434153, "Miguel Hidalgo": 414470, "Venustiano Carranza": 442798, "Xochimilco": 442178, "Tláhuac": 392313, "Cuajimalpa de Morelos": 217686, "La Magdalena Contreras": 247622, "Milpa Alta": 152685, } CATEGORIES = ["Gimnasios y espacios deportivos", "Servicios para personas adultas mayores"] HORIZONS_YEARS = [1, 3, 5] # t=0 fijado en la primera edición (nov-2022); años transcurridos reales # (no asumimos espaciado uniforme entre ediciones). EDITION_YEARS = { "2022-11": 0 / 12, "2023-11": 12 / 12, "2024-11": 24 / 12, "2025-05": 30 / 12, "2026-05": 42 / 12, } def load_all_editions() -> pd.DataFrame: frames = [] for folder, fname in EDITIONS: path = DATA_DIR / folder / fname cols = [ "edicion_denue", "anio_denue", "mes_denue", "categoria_proyecto", "subcategoria_proyecto", "municipio", "latitud", "longitud", "revision_manual", ] df = pd.read_csv(path, usecols=cols) frames.append(df) full = pd.concat(frames, ignore_index=True) full["years_elapsed"] = full["edicion_denue"].map(EDITION_YEARS) return full def ols_fit(x: np.ndarray, y: np.ndarray): """Regresión lineal simple con error estándar de predicción. Devuelve dict con pendiente, intercepto, r2 y una función predict(x0) -> (y_hat, intervalo_95).""" n = len(x) x_mean, y_mean = x.mean(), y.mean() sxx = np.sum((x - x_mean) ** 2) if sxx == 0 or n < 3: slope, intercept = 0.0, y_mean resid_se = 0.0 else: slope = np.sum((x - x_mean) * (y - y_mean)) / sxx intercept = y_mean - slope * x_mean y_hat = intercept + slope * x resid = y - y_hat dof = max(n - 2, 1) resid_se = float(np.sqrt(np.sum(resid ** 2) / dof)) y_hat_full = intercept + slope * x ss_res = np.sum((y - y_hat_full) ** 2) ss_tot = np.sum((y - y_mean) ** 2) r2 = 1 - ss_res / ss_tot if ss_tot > 0 else 0.0 def predict(x0: float): y0 = intercept + slope * x0 if sxx == 0 or n < 3: se = resid_se else: se = resid_se * np.sqrt(1 + 1 / n + (x0 - x_mean) ** 2 / sxx) # ~95% con z=1.96 (aproximación normal; n es pequeño, se trata # como una banda indicativa, no una garantía formal). return max(y0, 0.0), 1.96 * se return { "slope_per_year": round(float(slope), 3), "r2": round(float(r2), 3), "predict": predict, } def classify_semaforo(percapita_percentile: float, slope: float, slope_percentile: float) -> str: """Proxy de semáforo SOLO-OFERTA (ver limitación en el docstring del módulo): compara nivel de oferta per cápita contra el resto de alcaldías (terciles) y la tendencia de crecimiento reciente. """ oferta_alta = percapita_percentile >= 0.66 oferta_baja = percapita_percentile <= 0.33 crece = slope_percentile >= 0.66 or slope > 0 estable_o_baja = slope_percentile <= 0.33 or slope <= 0 if oferta_alta and estable_o_baja: return "rojo" if oferta_baja and (crece or True): # baja oferta relativa siempre es al menos zona a vigilar; # se vuelve verde franco si además no está saturándose rápido return "verde" return "amarillo" def compute_centroids(raw: pd.DataFrame) -> dict: """Centroide (promedio de lat/lon de los establecimientos) por alcaldía, usando la edición más reciente. Es un proxy razonable de dónde se concentra la actividad relevante para el reto dentro de cada alcaldía; no es el centroide administrativo oficial.""" last = raw[raw.edicion_denue == "2026-05"] out = {} for municipio, sub in last.groupby("municipio"): out[municipio] = { "lat": round(float(sub["latitud"].mean()), 5), "lon": round(float(sub["longitud"].mean()), 5), } return out def compute_outline(raw: pd.DataFrame) -> list: """Envolvente convexa de todos los puntos DENUE observados (todas las ediciones y categorías). Sirve como contorno de referencia muy aproximado de la zona urbana cubierta por los datos -- no es un límite administrativo real, sólo evita depender de un basemap externo en el prototipo v0.""" from scipy.spatial import ConvexHull pts = raw[["longitud", "latitud"]].dropna().to_numpy() hull = ConvexHull(pts) ring = pts[hull.vertices] return [[round(float(lon), 5), round(float(lat), 5)] for lon, lat in ring] def build_dataset() -> dict: raw = load_all_editions() centroids = compute_centroids(raw) outline = compute_outline(raw) agg = ( raw.groupby(["municipio", "categoria_proyecto", "edicion_denue", "years_elapsed"]) .size() .reset_index(name="conteo") .sort_values(["municipio", "categoria_proyecto", "years_elapsed"]) ) alcaldias = sorted(POBLACION_2020_ALCALDIA.keys()) last_edition = "2026-05" last_years = EDITION_YEARS[last_edition] zones = [] # Para calcular terciles necesitamos primero todas las métricas crudas. raw_metrics = [] # (municipio, categoria, percapita_actual, slope) fits_cache = {} for municipio in alcaldias: poblacion = POBLACION_2020_ALCALDIA[municipio] for categoria in CATEGORIES: sub = agg[(agg.municipio == municipio) & (agg.categoria_proyecto == categoria)] series = {row.edicion_denue: int(row.conteo) for row in sub.itertuples()} # completar ediciones sin registros con 0 for ed in EDITION_YEARS: series.setdefault(ed, 0) x = np.array([EDITION_YEARS[ed] for ed in EDITION_YEARS]) y = np.array([series[ed] for ed in EDITION_YEARS], dtype=float) fit_full = ols_fit(x, y) conteo_actual = series[last_edition] percapita = conteo_actual / poblacion * 100000 fits_cache[(municipio, categoria)] = (fit_full, series, percapita, poblacion) raw_metrics.append((municipio, categoria, percapita, fit_full["slope_per_year"])) percapita_vals = {} slope_vals = {} for categoria in CATEGORIES: pcs = sorted(v[2] for v in raw_metrics if v[1] == categoria) sls = sorted(v[3] for v in raw_metrics if v[1] == categoria) percapita_vals[categoria] = pcs slope_vals[categoria] = sls def percentile_rank(value, sorted_vals): if not sorted_vals: return 0.5 import bisect i = bisect.bisect_left(sorted_vals, value) return i / max(len(sorted_vals) - 1, 1) for municipio in alcaldias: for categoria in CATEGORIES: fit_full, series, percapita, poblacion = fits_cache[(municipio, categoria)] # --- validación retrospectiva: entrena solo con 2022-2024 --- train_eds = ["2022-11", "2023-11", "2024-11"] x_train = np.array([EDITION_YEARS[e] for e in train_eds]) y_train = np.array([series[e] for e in train_eds], dtype=float) fit_train = ols_fit(x_train, y_train) validation = [] for test_ed in ["2025-05", "2026-05"]: y_pred, band = fit_train["predict"](EDITION_YEARS[test_ed]) y_real = series[test_ed] err_abs = abs(y_pred - y_real) err_pct = (err_abs / y_real * 100) if y_real > 0 else (0.0 if y_pred == 0 else None) validation.append({ "edicion": test_ed, "predicho": round(y_pred, 1), "observado": y_real, "error_abs": round(err_abs, 1), "error_pct": round(err_pct, 1) if err_pct is not None else None, }) # --- proyección a futuro con todos los puntos reales --- proyecciones = {} for h in HORIZONS_YEARS: y_pred, band = fit_full["predict"](last_years + h) proyecciones[str(h)] = { "estimado": round(y_pred, 1), "banda_95": round(band, 1), } pc_rank = percentile_rank(percapita, percapita_vals[categoria]) sl_rank = percentile_rank(fit_full["slope_per_year"], slope_vals[categoria]) semaforo = classify_semaforo(pc_rank, fit_full["slope_per_year"], sl_rank) zones.append({ "alcaldia": municipio, "categoria": categoria, "poblacion_2020": poblacion, "centroide": centroids.get(municipio, {"lat": 19.4326, "lon": -99.1332}), "serie_historica": [ {"edicion": ed, "conteo": series[ed]} for ed in EDITION_YEARS ], "oferta_percapita_100k": round(percapita, 2), "oferta_percapita_percentil": round(pc_rank, 2), "tendencia_pendiente_anual": fit_full["slope_per_year"], "tendencia_percentil": round(sl_rank, 2), "tendencia_r2": fit_full["r2"], "proyeccion": proyecciones, "validacion_retrospectiva": validation, "semaforo": semaforo, }) return { "generado": "prototipo v0 - solo oferta DENUE, sin cruce demográfico ni de espacio público", "ediciones": list(EDITION_YEARS.keys()), "categorias": CATEGORIES, "contorno_referencia": outline, "zonas": zones, } if __name__ == "__main__": dataset = build_dataset() out_path = Path(__file__).resolve().parent.parent / "data" / "semaforo_v0.json" out_path.write_text(json.dumps(dataset, ensure_ascii=False, indent=2), encoding="utf-8") print(f"OK -> {out_path} ({len(dataset['zonas'])} filas alcaldía x categoría)")
